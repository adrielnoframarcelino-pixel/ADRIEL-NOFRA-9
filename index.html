1. Builder Pattern - Magma Chamber Creational

Builder Pattern seperti magma yang naik bertahap sebelum erupsi. Membangun objek kompleks secara bertahap.

📁 magma_builder.dart
// ==========================================
// Nama  : ADRIEL NOFRA MARCELINO
// NIM   : 251410017
// Volcano Theme - API Volkanik
// ==========================================

class EruptionConfig {}

class EruptionBuilder {
  EruptionBuilder lavaTemp(int celcius) {
    print("🌡️ Suhu Lava : $celcius °C");
    return this;
  }
  EruptionBuilder ashHeight(double km) {
    print("💨 Abu Vulkanik : $km km");
    return this;
  }
  EruptionBuilder magmaType(String type) {
    print("🗻 Tipe Magma : $type");
    return this;
  }
  Eruption build() {
    print("🌋 GUNUNG MERAPI SIAP MEletus!");
    return Eruption();
  }
}

void main() {
  EruptionBuilder()
      .lavaTemp(1100)
      .ashHeight(15)
      .magmaType('Basaltik')
      .build();
}

3. Extension Methods - LETUSAN ! Dengan Erupsi

Extension menambahkan method "meletus" langsung ke tipe data String, layaknya API Volkanik.

📁 eruption_extension.dart
// ==========================================
// Nama  : ADRIEL NOFRA MARCELINO
// NIM   : 251410017
// ==========================================

extension VolcanicExtension on String {
  String erupt() => "🌋 $this MELEDAK! 🔥";
  bool get isVolcanoName => contains('Gunung') || contains('Merapi');
  String toLavaCase() => toUpperCase() + "!!! LAVA!!!";
}

void main() {
  var gunung = 'Gunung Semeru';
  print(gunung.erupt());
  print("Apakah nama gunung? ${gunung.isVolcanoName}");
  print('magma'.toLavaCase());
}

extension VolcanicIndex on int {
  String get skalaErupsi {
    if (this >= 7) return "☠️ SUPERKOLOSAL (VEI $this)";
    if (this >= 4) return "🌋 BESAR (VEI $this)";
    return "🔥 SEDANG (VEI $this)";

}

void main() {
  int vei = 5;
  print(vei.skalaErupsi);
  print("Berbahaya? ${vei.isDangerous}");
}

  bool get isErupting => this != null && this!.isNotEmpty;

14. Media Volcano Challenge 🌋 Polymorphism API

Implementasi polimorfisme untuk berbagai jenis pemantau gunung api (Sensor thermal, seismik, visual).

📁 media_volcano_api.dart
// ==========================================
// Nama  : ADRIEL NOFRA MARCELINO
// NIM   : 251410017
// Volcano Media Monitoring
// ==========================================

abstract class VolcanoMonitor {
  void monitor();
}

class ThermalSensor implements VolcanoMonitor {
  @override void monitor() => print("🌡️ Sensor Thermal: Suhu magma naik!");
}
class SeismicSensor implements VolcanoMonitor {
  @override void monitor() => print("📊 Sensor Seismik: Gempa vulkanik terdeteksi!");
}
class VisualCamera implements VolcanoMonitor {
  @override void monitor() => print("📹 Kamera Visual: Asap mengepul dari kawah!");
}

class VolcanoAlertSystem {
  List sensors = [];
  void addSensor(VolcanoMonitor s) => sensors.add(s);
  void triggerAll() => sensors.forEach((s) => s.monitor());
}

void main() {
  var alert = VolcanoAlertSystem();
  alert.addSensor(ThermalSensor());
  alert.addSensor(SeismicSensor());
  alert.addSensor(VisualCamera());
  print("=== SISTEM PERINGATAN DINI GUNUNG API ===");
  alert.triggerAll();
}
